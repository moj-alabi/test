Keep-Alive Script

Run this on each device to periodically ping the server and keep it marked as active.

#!/bin/bash
# Run on remote device to keep it marked as online
SERVER="http://YOUR_SERVER_IP:5000"
ID="worker-01"

while true; do
  curl -s -X POST "$SERVER/api/bots/ping" \
    -H "Content-Type: application/json" \
    -d "{\"id\":\"$ID\"}" > /dev/null
  sleep 30
done
